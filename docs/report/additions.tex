 		.1 .

 		.3 РПЗ.pdf.
 		.2 code\DTcomment{исходный код проекта}.
 		.3 docker \ldots{} \DTcomment{файлы для развертывания контейнера}.
 		.3 sql \ldots{} \DTcomment{SQL скрипты}.
 		.3 gradle \ldots{} \DTcomment{дополнительные скрипты системы сборки}.
 		.3 layers\DTcomment{слои чистой архитектуры}.

 		.4 business-logic \ldots{} \DTcomment{исходный код бизнес логики}.
 		.4 ui \ldots{} \DTcomment{исходный код контроллеров}.
 		.3 src \ldots{} \DTcomment{исходный код точки входа в приложение}.
 		.3 gradle.build\DTcomment{главный скрипт системы сборки}.
 		.3 settings.gradle\DTcomment{конфигурационный файл сборки}.
 		}
 \end{enumerate}
